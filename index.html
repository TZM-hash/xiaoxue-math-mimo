建表 SQL（首次配置需执行）

在 Supabase 控制台 → SQL Editor 中执行：

CREATE TABLE user_data (
  owner_id text NOT NULL,
  device_id text NOT NULL,
  profiles jsonb NOT NULL DEFAULT '[]',
  active_id text DEFAULT '',
  updated_at timestamptz DEFAULT now(),
  PRIMARY KEY (owner_id, device_id)
);

CREATE TABLE user_settings (
  owner_id text NOT NULL,
  device_id text NOT NULL,
  settings jsonb NOT NULL DEFAULT '{}',
  updated_at timestamptz DEFAULT now(),
  PRIMARY KEY (owner_id, device_id)
);

ALTER TABLE user_data ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_settings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "allow_all_user_data" ON user_data FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all_user_settings" ON user_settings FOR ALL USING (true) WITH CHECK (true);
